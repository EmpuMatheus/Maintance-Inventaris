# ============================================================

# OFFICE INVENTORY MAINTENANCE SYSTEM

# Environment Variables Example

# ============================================================

#

# IMPORTANT:

#

# 1. Never place real production secrets in this file.

#

# ↓

# .env

#

# ============================================================

# ============================================================

# APPLICATION

# ============================================================

NODE_ENV=development

APP_NAME=Office Inventory Maintenance System

APP_VERSION=1.0.0

# ============================================================

# BACKEND SERVER

# ============================================================

PORT=3000

API_PREFIX=/api/v1

# ============================================================

# FRONTEND

# ============================================================

VITE_APP_NAME=Office Inventory Maintenance System

VITE_API_URL=http://localhost:3000/api/v1

# ============================================================

# DATABASE

# ============================================================

# Format:

#

# postgresql://USERNAME:PASSWORD@HOST:PORT/DATABASE

#

# Example only:

#

# postgresql://postgres:password@localhost:5432/office_inventory

#

# Replace with your own local credentials.

DATABASE_URL=postgresql://USERNAME:PASSWORD@localhost:5432/office_inventory

# ============================================================

# DATABASE POOL

# ============================================================

DATABASE_POOL_MIN=2

DATABASE_POOL_MAX=10

DATABASE_CONNECTION_TIMEOUT_MS=10000

# ============================================================

# AUTHENTICATION

# ============================================================

# Generate a strong random secret for actual environments.

#

# DO NOT use this placeholder value in production.

JWT_SECRET=CHANGE_ME_WITH_A_STRONG_RANDOM_SECRET

JWT_EXPIRES_IN=15m

# ============================================================

# REFRESH TOKEN

# ============================================================

# Enable these when refresh-token authentication is implemented.

JWT_REFRESH_SECRET=CHANGE_ME_WITH_ANOTHER_STRONG_RANDOM_SECRET

JWT_REFRESH_EXPIRES_IN=7d

# ============================================================

# PASSWORD HASHING

# ============================================================

# Algorithm used by the backend.

#

# Recommended:

#

# argon2id

PASSWORD_HASH_ALGORITHM=argon2id

# ============================================================

# CORS

# ============================================================

# Development frontend.

CORS_ORIGIN=http://localhost:5173

# ============================================================

# RATE LIMIT

# ============================================================

RATE_LIMIT_WINDOW_MS=900000

RATE_LIMIT_MAX=500

# ============================================================

# AUTH RATE LIMIT

# ============================================================

AUTH_RATE_LIMIT_WINDOW_MS=900000

AUTH_RATE_LIMIT_MAX=10

# ============================================================

# REQUEST BODY

# ============================================================

REQUEST_BODY_LIMIT=2mb

# ============================================================

# FILE STORAGE

# ============================================================

# Development can use local storage.

#

# Production should use persistent storage.

STORAGE_DRIVER=local

STORAGE_PATH=./storage

# ============================================================

# FILE UPLOAD

# ============================================================

# Maximum image upload:

#

# 5 MB

UPLOAD_IMAGE_MAX_SIZE=5242880

# Maximum document upload:

#

# 10 MB

UPLOAD_DOCUMENT_MAX_SIZE=10485760

# ============================================================

# ALLOWED IMAGE TYPES

# ============================================================

# ============================================================

# ALLOWED DOCUMENT TYPES

# ============================================================

UPLOAD_ALLOWED_DOCUMENT_TYPES=application/pdf

# ============================================================

# LOGGING

# ============================================================

# Available examples:

#

# debug

# info

# warn

# error

LOG_LEVEL=debug

# ============================================================

# PAGINATION

# ============================================================

MAX_PAGE_SIZE=100

# ============================================================

# MAINTENANCE

# ============================================================

# ============================================================

# QR

# ============================================================

# Current strategy:

#

# asset_code

#

# QR contains asset code and current asset data is retrieved

# from the API.

# ============================================================

# TIMEZONE

# ============================================================

# Application display/business timezone.

#

# Database timestamps should still follow the project's

# UTC storage strategy where applicable.

# LOCALE

# ============================================================
